## 一、WHERE 条件详解

### 比较运算符

```sql
-- 基础比较
WHERE age = 18
WHERE age != 18
WHERE age <> 18      -- 不等号，同 !=
WHERE age > 18
WHERE age >= 18
WHERE age BETWEEN 18 AND 35

-- 字符串比较
WHERE name = '张三'
WHERE name > '李'    -- 按字典序比较（UTF-8 编码顺序）
```

### 逻辑运算

```sql
-- AND（且）
WHERE age >= 18 AND status = 1

-- OR（或）
WHERE city = '北京' OR city = '上海'

-- NOT（非）或 <>
WHERE NOT status = 0
WHERE status <> 0

-- 优先级：NOT > AND > OR
-- 建议用括号明确优先级：
WHERE (status = 1 OR level = 'vip') AND age >= 18
```

### IN 与 NOT IN

```sql
-- IN 列表
WHERE city IN ('北京', '上海', '广州')
WHERE city NOT IN ('北京', '上海')

-- IN 子查询（注意性能，大数据量时 EXISTS 更优）
WHERE user_id IN (SELECT id FROM vip_users)
```

### LIKE 模糊查询

```sql
WHERE name LIKE '张%'         -- 以"张"开头
WHERE name LIKE '%张'         -- 以"张"结尾
WHERE name LIKE '%张%'        -- 包含"张"
WHERE name LIKE '张_'         -- "张" + 任意一个字符
WHERE name LIKE '张__'        -- "张" + 任意两个字符
```

### NULL 判断

```sql
WHERE phone IS NULL           -- 是 NULL
WHERE phone IS NOT NULL       -- 不是 NULL

-- ⚠️ 错误写法：
WHERE phone = NULL     -- 永远不成立！NULL 不能用于 = 比较
WHERE phone != NULL    -- 也永远不成立
```

### EXISTS

```sql
-- 查询有订单的用户
SELECT * FROM users u
WHERE EXISTS (
    SELECT 1 FROM orders o WHERE o.user_id = u.id
);

-- 查询没有订单的用户
SELECT * FROM users u
WHERE NOT EXISTS (
    SELECT 1 FROM orders o WHERE o.user_id = u.id
);
```

`EXISTS` 与 `IN` 的选择：
- 子查询结果集小 → `IN` 可能更快
- 外层表小、子查询大 → `EXISTS` 更优
- 可以用 `JOIN` 替代的场景优先用 `JOIN`

## 二、ORDER BY 排序

### 单字段排序

```sql
SELECT * FROM products ORDER BY price;          -- 升序（默认）
SELECT * FROM products ORDER BY price ASC;      -- 升序（显式）
SELECT * FROM products ORDER BY price DESC;     -- 降序
```

### 多字段排序

```sql
-- 先按分类升序，再按价格降序
SELECT * FROM products ORDER BY category_id ASC, price DESC;
```

### 表达式排序

```sql
-- 按折扣价排序
SELECT *, price * discount AS final_price
FROM products
ORDER BY final_price;

-- 按长度排序
SELECT * FROM articles ORDER BY LENGTH(content) DESC;

-- 随机排序（大数据量时极慢，慎用）
SELECT * FROM users ORDER BY RAND() LIMIT 10;
```

### 排序与索引

```sql
-- 如果排序字段有索引，排序很快
-- 如果 join 中有排序，排序通常发生在 join 之后
-- 在金额、时间等经常排序的字段上建索引
```

## 三、GROUP BY 分组

### 基础分组

```sql
-- 按状态分组统计
SELECT status, COUNT(*) AS cnt
FROM orders
GROUP BY status;

-- 按年份分组
SELECT YEAR(created_at) AS year, SUM(total_amount) AS total
FROM orders
GROUP BY YEAR(created_at);
```

### 多字段分组

```sql
-- 按年份和月份分组
SELECT
    YEAR(created_at) AS year,
    MONTH(created_at) AS month,
    COUNT(*) AS order_count,
    SUM(total_amount) AS revenue
FROM orders
GROUP BY YEAR(created_at), MONTH(created_at)
ORDER BY year DESC, month DESC;
```

### HAVING 过滤分组

```sql
-- WHERE 在分组前过滤，HAVING 在分组后过滤
SELECT user_id, COUNT(*) AS order_count, SUM(total_amount) AS total
FROM orders
WHERE status = 1               -- 只统计已支付的订单
GROUP BY user_id
HAVING order_count >= 5        -- 只显示下单 >= 5 的用户
   AND total > 1000;            -- 且总金额 > 1000
```

### GROUP_CONCAT

```sql
-- 将分组内的值拼接为字符串
SELECT
    user_id,
    GROUP_CONCAT(order_no ORDER BY created_at DESC SEPARATOR ', ') AS orders
FROM orders
GROUP BY user_id;
```

## 四、LIMIT 分页

### 基础分页

```sql
-- LIMIT 偏移量, 条数
SELECT * FROM users ORDER BY id LIMIT 0, 20;     -- 第1页（1-20条）
SELECT * FROM users ORDER BY id LIMIT 20, 20;    -- 第2页（21-40条）
SELECT * FROM users ORDER BY id LIMIT 40, 20;    -- 第3页（41-60条）

-- 另一种写法
SELECT * FROM users ORDER BY id LIMIT 20 OFFSET 40;
```

### 深分页优化

```sql
-- ❌ 当偏移量很大时，MySQL 需要扫描并丢弃前 100000 行
SELECT * FROM users ORDER BY id LIMIT 100000, 20;

-- ✅ 方案1：基于游标（推荐）
SELECT * FROM users WHERE id > 100000 ORDER BY id LIMIT 20;

-- ✅ 方案2：子查询优化
SELECT * FROM users
WHERE id >= (SELECT id FROM users ORDER BY id LIMIT 100000, 1)
ORDER BY id LIMIT 20;

-- ✅ 方案3：JOIN 优化
SELECT u.* FROM users u
JOIN (SELECT id FROM users ORDER BY id LIMIT 100000, 20) AS tmp
ON u.id = tmp.id;
```

### TOP-N 查询

```sql
-- 每个分类下价格最高的商品
SELECT * FROM (
    SELECT *,
        ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY price DESC) AS rn
    FROM products
) t WHERE rn = 1;
```

## 五、条件表达式

### CASE WHEN

```sql
SELECT
    name,
    CASE
        WHEN age < 18 THEN '少年'
        WHEN age < 35 THEN '青年'
        WHEN age < 60 THEN '中年'
        ELSE '老年'
    END AS age_group
FROM users;

-- 简单 CASE（等值判断）
SELECT
    status,
    CASE status
        WHEN 0 THEN '待支付'
        WHEN 1 THEN '已支付'
        WHEN 2 THEN '已发货'
        WHEN 3 THEN '已完成'
        ELSE '未知'
    END AS status_text
FROM orders;
```

### IF 函数

```sql
SELECT name, IF(age >= 18, '成年', '未成年') AS adult FROM users;

-- IFNULL（NULL 处理）
SELECT name, IFNULL(phone, '未填写') AS phone FROM users;

-- COALESCE（返回第一个非 NULL 值）
SELECT COALESCE(phone, email, '无联系方式') AS contact FROM users;
```

## 六、聚合函数进阶

```sql
-- 常用聚合
COUNT(*)        -- 总行数
COUNT(col)      -- 非 NULL 行数
COUNT(DISTINCT col)  -- 去重后的数量

SUM(col)        -- 求和
AVG(col)        -- 平均值

MAX(col)        -- 最大值
MIN(col)        -- 最小值

GROUP_CONCAT(col)  -- 字符串拼接

-- 汇总统计（WITH ROLLUP）
SELECT
    IFNULL(status, '合计') AS status,
    COUNT(*) AS cnt,
    SUM(total_amount) AS total
FROM orders
GROUP BY status WITH ROLLUP;
```

## 七、窗口函数（MySQL 8.0+）

### 排名函数

```sql
SELECT
    name,
    salary,
    ROW_NUMBER() OVER (ORDER BY salary DESC) AS ranking,         -- 无并列
    RANK() OVER (ORDER BY salary DESC) AS rank_with_gap,         -- 有并列有间隔
    DENSE_RANK() OVER (ORDER BY salary DESC) AS rank_no_gap      -- 有并列无间隔
FROM employees;
```

### 聚合窗口

```sql
SELECT
    DATE(created_at) AS day,
    SUM(amount) AS daily_total,
    SUM(SUM(amount)) OVER (ORDER BY DATE(created_at)) AS running_total
FROM orders
GROUP BY DATE(created_at);
```

### 分区窗口

```sql
-- 每组内排名
SELECT
    category_id,
    name,
    price,
    ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY price DESC) AS ranking
FROM products;
```

## 八、实用查询模式

### 去重查询

```sql
-- 方法1：DISTINCT
SELECT DISTINCT category_id FROM products;

-- 方法2：GROUP BY
SELECT category_id FROM products GROUP BY category_id;

-- 方法3：窗口函数（取每组最新一条）
SELECT * FROM (
    SELECT *,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY created_at DESC) AS rn
    FROM orders
) t WHERE rn = 1;
```

### 统计占比

```sql
SELECT
    status,
    COUNT(*) AS cnt,
    COUNT(*) * 100.0 / SUM(COUNT(*)) OVER() AS percentage
FROM orders
GROUP BY status;
```

### 连续登录用户

```sql
-- 假设有 login_logs(user_id, login_date)
SELECT DISTINCT user_id
FROM (
    SELECT
        user_id,
        login_date,
        DATE_SUB(login_date, INTERVAL ROW_NUMBER() OVER (
            PARTITION BY user_id ORDER BY login_date
        ) DAY) AS group_date
    FROM login_logs
) t
GROUP BY user_id, group_date
HAVING COUNT(*) >= 3;
```
