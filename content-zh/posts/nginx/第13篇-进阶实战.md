## 场景一：SPA 单页应用

Vue/React 构建的单页应用，核心需求是：除静态资源外所有路径返回 `index.html`。

```nginx
server {
    listen 80;
    server_name app.example.com;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name app.example.com;

    root /var/www/app/dist;

    # SSL
    ssl_certificate /etc/letsencrypt/live/app.example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/app.example.com/privkey.pem;

    # 静态资源（hash 文件名，永久缓存）
    location /assets/ {
        expires 365d;
        add_header Cache-Control "public, immutable";
        access_log off;
    }

    # 其他路径返回 index.html（SPA 路由）
    location / {
        try_files $uri $uri/ /index.html;

        # HTML 不缓存
        add_header Cache-Control "no-cache, must-revalidate";
    }

    # 禁止访问隐藏文件
    location ~ /\. {
        deny all;
        return 404;
    }
}
```

## 场景二：HTTPS 反向代理

带 SSL 终结的反向代理，后端服务不需要处理证书。

```nginx
upstream backend {
    least_conn;
    keepalive 32;
    server 10.0.0.1:8080 max_fails=3 fail_timeout=30s;
    server 10.0.0.2:8080 max_fails=3 fail_timeout=30s;
}

server {
    listen 80;
    server_name api.example.com;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name api.example.com;

    ssl_certificate /etc/letsencrypt/live/api.example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.example.com/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_session_cache shared:SSL:10m;

    location / {
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout 5s;
        proxy_read_timeout 30s;
        proxy_send_timeout 30s;

        proxy_next_upstream error timeout http_500;
        proxy_next_upstream_tries 2;
    }
}
```

## 场景三：文件上传服务

```nginx
server {
    listen 80;
    server_name upload.example.com;

    client_max_body_size 500m;         # 允许上传最大 500MB
    client_body_buffer_size 2m;        # 超过 2MB 写入临时文件
    client_body_timeout 300s;          # 上传超时 5 分钟

    proxy_connect_timeout 60s;
    proxy_read_timeout 300s;
    proxy_send_timeout 300s;

    location / {
        proxy_pass http://upload_backend:5000;
        proxy_set_header Host $host;
    }
}
```

## 场景四：WebSocket 代理

```nginx
map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}

upstream ws_backend {
    server 10.0.0.1:3000;
    server 10.0.0.2:3000;
}

server {
    listen 80;
    server_name ws.example.com;

    location /ws {
        proxy_pass http://ws_backend;

        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;

        # WebSocket 长连接超时
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }
}
```

## 场景五：WordPress 站点

```nginx
upstream php {
    server unix:/var/run/php/php8.3-fpm.sock;
}

server {
    listen 80;
    server_name blog.example.com;

    root /var/www/wordpress;
    index index.php;

    # 静态资源直接返回
    location ~* \.(css|js|jpg|jpeg|png|gif|ico|woff|woff2|ttf|svg)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
        access_log off;
        log_not_found off;
    }

    # WP 内置文件
    location = /favicon.ico {
        log_not_found off;
        access_log off;
    }

    location = /robots.txt {
        log_not_found off;
        access_log off;
    }

    # Deny hidden files
    location ~ /\. {
        deny all;
    }

    # WordPress 路由
    location / {
        try_files $uri $uri/ /index.php?$args;
    }

    # PHP 处理
    location ~ \.php$ {
        fastcgi_pass php;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;

        fastcgi_buffer_size 128k;
        fastcgi_buffers 4 256k;
        fastcgi_busy_buffers_size 256k;
    }

    # 禁止执行某些目录下的 PHP
    location ~* /(wp-content/uploads|wp-content/cache)/.*\.php$ {
        deny all;
    }
}
```

## 场景七：多版本 API 网关

```nginx
upstream v1_backend {
    server 10.0.0.1:8080;
}

upstream v2_backend {
    server 10.0.0.2:8080;
}

upstream v3_canary {
    server 10.0.0.3:8080 weight=1;
    server 10.0.0.2:8080 weight=99;  # 只有 1% 流量到新版本
}

server {
    listen 80;
    server_name api.example.com;

    # 版本路由
    location /v1/ {
        proxy_pass http://v1_backend/;
    }

    location /v2/ {
        proxy_pass http://v2_backend/;
    }

    location /v3/ {
        proxy_pass http://v3_canary/;
    }

    # 无版本号默认走最新稳定版
    location / {
        proxy_pass http://v2_backend/;
    }

    # 灰度发布：根据 Header 切换版本
    location / {
        if ($http_x_api_version = "v3") {
            proxy_pass http://v3_canary/;
            break;
        }
        proxy_pass http://v2_backend/;
    }
}
```

## 场景八：下载站点限速

```nginx
server {
    listen 80;
    server_name download.example.com;

    root /var/www/downloads;

    # 全局限速
    limit_rate 200k;            # 单连接 200KB/s
    limit_rate_after 10m;       # 前 10MB 不限速（快速开始）

    # 不同文件类型不同限速
    location ~* \.(zip|tar\.gz)$ {
        limit_rate 500k;
    }

    location ~* \.(iso|dmg)$ {
        limit_rate 2m;          # ISO 文件 2MB/s
        limit_rate_after 100m;
    }

    # 会员不限速
    location /vip/ {
        limit_rate 0;           # 0 表示不限速
        valid_referers server_names ~\.example\.com;
        if ($invalid_referer) {
            limit_rate 100k;
        }
    }

    # 开启目录列表
    location / {
        autoindex on;
        autoindex_exact_size off;
        autoindex_localtime on;
    }

    # 限制并发连接
    limit_conn per_ip 5;
    limit_conn_status 503;
}
```

## 场景九：PHP-FPM 优化配置

```nginx
upstream php_backend {
    server unix:/var/run/php/php8.3-fpm.sock;
    server unix:/var/run/php/php8.3-fpm-2.sock;
}

server {
    location ~ \.php$ {
        fastcgi_pass php_backend;
        fastcgi_index index.php;
        include fastcgi_params;

        # PHP-FPM 参数
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param PHP_VALUE "max_execution_time=300\nmax_input_time=300";

        # 缓冲优化
        fastcgi_buffer_size 128k;
        fastcgi_buffers 4 256k;
        fastcgi_busy_buffers_size 256k;
        fastcgi_temp_file_write_size 256k;

        # 超时
        fastcgi_connect_timeout 30s;
        fastcgi_read_timeout 300s;
        fastcgi_send_timeout 300s;

        # 缓存
        fastcgi_cache my_cache;
        fastcgi_cache_valid 200 302 10m;
        fastcgi_cache_use_stale error timeout updating;
    }
}
```

## 场景十：CORS 跨域配置

```nginx
# 通用 CORS 配置（可放入 snippets/cors.conf）
set $cors_origin "";
if ($http_origin ~* (https?://.*\.example\.com$)) {
    set $cors_origin $http_origin;
}

add_header Access-Control-Allow-Origin $cors_origin always;
add_header Access-Control-Allow-Methods "GET, POST, PUT, DELETE, PATCH, OPTIONS" always;
add_header Access-Control-Allow-Headers "Content-Type, Authorization, X-Requested-With" always;
add_header Access-Control-Allow-Credentials "true" always;
add_header Access-Control-Max-Age "86400" always;

# 预检请求
if ($request_method = OPTIONS) {
    return 204;
}
```

```nginx
}
```

## 场景十一：国际化多站点

```nginx
map $accept_language $lang {
    default en;
    ~^zh zh;
    ~^ja ja;
    ~^ko ko;
}

server {
    listen 80;
    server_name example.com;

    # 根据 Accept-Language 重定向
    location / {
        if ($lang = zh) {
            rewrite ^ /zh$uri last;
        }
        if ($lang = ja) {
            rewrite ^ /ja$uri last;
        }
    }

    location /en {
        alias /var/www/example/en;
        index index.html;
    }

    location /zh {
        alias /var/www/example/zh;
        index index.html;
    }

    location /ja {
        alias /var/www/example/ja;
        index index.html;
    }
}

# 子域名方案
server {
    listen 80;
    server_name en.example.com;
    root /var/www/example/en;
}

server {
    listen 80;
    server_name cn.example.com;
    root /var/www/example/zh;
}
```

## 场景十二：临时维护页面

```nginx
# 方法一：503 维护页面
server {
    listen 80;
    server_name example.com;

    # 如果存在维护标记文件，显示维护页面
    if (-f $document_root/maintenance.enable) {
        return 503;
    }

    location / {
        proxy_pass http://backend;
    }

    error_page 503 @maintenance;
    location @maintenance {
        root /var/www/errors;
        rewrite ^(.*)$ /maintenance.html break;
    }
}

# 方法二：白名单（内网和指定 IP 绕过维护页面）
geo $maintenance_bypass {
    default 0;
    192.168.1.0/24 1;
    10.0.0.0/8 1;
    114.114.114.114 1;  # 指定测试 IP
}

server {
    if (-f $document_root/maintenance.enable && $maintenance_bypass = 0) {
        return 503;
    }

    # ... 正常配置
}
```

## 遇到问题怎么办？

虽然系列结束了，但实际使用中总会遇到新问题。这里是一些通用的排查思路：

```bash
# 1. 检查配置语法
nginx -t

# 2. 查看错误日志
tail -f /var/log/nginx/error.log

# 3. 查看访问日志（最近 100 行）
tail -100 /var/log/nginx/access.log

# 4. 查看进程状态
ps aux | grep nginx

# 5. 测试后端连通性
curl http://127.0.0.1:8080/health

# 6. 检查端口监听
ss -tlnp | grep nginx

# 7. 检查防火墙
sudo iptables -L -n | grep 80
```
